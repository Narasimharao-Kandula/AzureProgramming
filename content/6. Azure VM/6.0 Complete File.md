```powershell
Write-Host "===== EXPERIMENT 6 READINESS CHECK =====" -ForegroundColor Cyan

$os = Get-CimInstance Win32_OperatingSystem
$cpu = Get-CimInstance Win32_Processor
$totalRAM = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)
$freeRAM = [math]::Round($os.FreePhysicalMemory / 1MB, 2)
$disk = Get-CimInstance Win32_LogicalDisk -Filter "DeviceID='C:'"

Write-Host "[OS]      $($os.Caption) $($os.OSArchitecture)"

Write-Host "[RAM]     Total: ${totalRAM} GB | Free: ${freeRAM} GB"

try { $ping = Test-Connection google.com -Count 2 -Quiet

Write-Host "[RDP]     $(If(Test-Path 'C:\Windows\System32\mstsc.exe'){'OK'}Else{'MISSING'})"
Write-Host "[BROWSER] $(If(Test-Path 'C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe'){'Edge OK'}Else{'Check browser'})"
```

```jsp
<%@ page language="java" contentType="text/html; charset=UTF-8" pageEncoding="UTF-8"%>
<%@ page import="java.sql.*" %>
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>Student Records</title>
</head>
<body style="font-family:Arial;padding:40px;background:#f0f0f0">
<div style="width:800px;margin:auto;background:#fff;padding:30px;border:1px solid #ccc">
    <h2 style="color:#2c5364;text-align:center">Student Database Records</h2>
<%
    String url  = "jdbc:sqlserver://localhost:1433;databaseName=StudentDB;encrypt=true;trustServerCertificate=true;";
    String user = "sa";
    String pass = "P@ssw0rd123!";

    String fn = request.getParameter("firstName");
    String ln = request.getParameter("lastName");
    String em = request.getParameter("email");
    String dp = request.getParameter("department");
    int total = 0;
    Connection conn = null;

    try {
        Class.forName("com.microsoft.sqlserver.jdbc.SQLServerDriver");
        conn = DriverManager.getConnection(url, user, pass);

        if (fn != null && !fn.trim().isEmpty()) {
            PreparedStatement ps = conn.prepareStatement(
                "INSERT INTO Students(FirstName,LastName,Email,Department) VALUES(?,?,?,?)");
            ps.setString(1, fn.trim());
            ps.setString(2, ln.trim());
            ps.setString(3, em.trim());
            ps.setString(4, dp.trim());
            ps.executeUpdate();
            ps.close();
%>
    <p style="color:green;text-align:center;font-weight:bold">
        Student "<%=fn%> <%=ln%>" registered successfully!
    </p>
<%
        }

        ResultSet cnt = conn.createStatement().executeQuery("SELECT COUNT(*) FROM Students");
        if (cnt.next()) total = cnt.getInt(1);
        cnt.close();
%>
    <p style="text-align:center">Total Students: <b><%=total%></b></p>
    <table border="1" cellpadding="10" cellspacing="0" style="width:100%;border-collapse:collapse">
        <tr style="background:#2c5364;color:#fff">
            <th>ID</th><th>First Name</th><th>Last Name</th><th>Email</th><th>Department</th>
        </tr>
<%
        ResultSet rs = conn.createStatement().executeQuery("SELECT * FROM Students ORDER BY StudentID DESC");
        while (rs.next()) {
%>
        <tr>
            <td><%=rs.getInt("StudentID")%></td>
            <td><%=rs.getString("FirstName")%></td>
            <td><%=rs.getString("LastName")%></td>
            <td><%=rs.getString("Email")%></td>
            <td><%=rs.getString("Department")%></td>
        </tr>
<%
        }
        rs.close();
    } catch (Exception e) {
%>
    <p style="color:red;text-align:center;font-weight:bold">Error: <%=e.getMessage()%></p>
<%
    } finally {
        if (conn != null) try { conn.close(); } catch (Exception x) {}
    }
%>
    </table>
    <br>
    <p style="text-align:center">
        <a href="index.jsp">Register Another</a> |
        <a href="displayStudents.jsp">Refresh Records</a>
    </p>
</div>
</body>
</html>
```
