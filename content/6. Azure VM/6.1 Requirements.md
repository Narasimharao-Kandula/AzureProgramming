# Step 1 — Complete Process Cleanup & RAM Free

**This script kills ALL non-essential processes regardless of what apps are installed.** Only critical Windows system processes are protected.

**Open PowerShell (Admin)**

**Paste this entire block → Press Enter:**

```powershell
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "  STEP 1: FULL PROCESS CLEANUP & RAM FREE" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan

$before = [math]::Round((Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory / 1MB, 2)
Write-Host "`nFree RAM BEFORE: $before GB" -ForegroundColor Yellow

# ══════════════════════════════════════════
# CRITICAL SYSTEM PROCESSES — NEVER KILL
# Only these survive. Everything else dies.
# ══════════════════════════════════════════
$protected = @(
    "System","Idle","smss","csrss","wininit","services","lsass","lsm",
    "svchost","dwm","winlogon","fontdrvhost","sihost","ctfmon",
    "explorer","RuntimeBroker","ShellExperienceHost","StartMenuExperienceHost",
    "SearchHost","SearchIndexer","SearchProtocolHost","SearchFilterHost",
    "TextInputHost","taskhostw","conhost","spoolsv","audiodg",
    "MsMpEng","NisSrv","SecurityHealthService","SecurityHealthSystray",
    "Memory Compression","Registry","Sppsvc","WmiPrvSE",
    "WUDFHost","dasHost","dllhost","msdtc","CompatTelRunner",
    "MpCmdRun","Idle","System Idle Process",
    "powershell","pwsh","WindowsTerminal","conhost",
    "obs-studio","obs64","obs32"
)

# ══════════════════════════════════════════
# PHASE A: Kill ALL non-system user processes
# ══════════════════════════════════════════
Write-Host "`n--- Phase A: Killing ALL non-system processes ---" -ForegroundColor Cyan
$killedCount = 0
$totalFreedMB = 0

Get-Process | ForEach-Object {
    if ($protected -notcontains $_.ProcessName -and $_.Id -ne 0 -and $_.Id -ne 4) {
        $mb = [math]::Round($_.WorkingSet64 / 1MB, 0)
        if ($mb -gt 5) {
            Write-Host "  [KILL] $($_.ProcessName) (PID: $($_.Id), $mb MB)" -ForegroundColor Red
            Stop-Process -Id $_.Id -Force -ErrorAction SilentlyContinue
            $killedCount++
            $totalFreedMB += $mb
        }
    }
}
Write-Host "  Killed $killedCount processes, freed ~$totalFreedMB MB" -ForegroundColor Green

# ══════════════════════════════════════════
# PHASE B: Stop ALL unnecessary Windows services
# ══════════════════════════════════════════
Write-Host "`n--- Phase B: Stopping unnecessary services ---" -ForegroundColor Cyan

$svcsToStop = @(
    "DiagTrack","dmwappushservice","SysMain","WSearch",
    "Fax","TabletInputService","TouchInputService",
    "MapsBroker","lfsvc","SharedAccess","RemoteRegistry",
    "WMPNetworkSvc","XblAuthManager","XblGameSave",
    "XboxNetApiSvc","XboxGipSvc","wisvc","WalletService",
    "RetailDemo","PhoneSvc","PcaSvc","CDPSvc",
    "CDPUserSvc","DsSvc","MessagingService",
    "UserDataSvc","UnistoreSvc","OneSyncSvc",
    "PrintNotify","Spooler","BTAGService","BthAvctpSvc",
    "BluetoothUserService","bthserv",
    "TabletInputService","TouchInputService",
    "WbioSrvc","WerSvc","DusmSvc",
    "InstallService","DoSvc","wuauserv",
    "BITS","TrkWks","EFS"
)

foreach ($s in $svcsToStop) {
    $svc = Get-Service -Name $s -ErrorAction SilentlyContinue
    if ($svc -and $svc.Status -eq "Running") {
        try {
            Stop-Service -Name $s -Force -ErrorAction Stop
            Write-Host "  [STOPPED] $s" -ForegroundColor Green
        } catch {
            Write-Host "  [SKIPPED] $s (protected by Windows)" -ForegroundColor DarkGray
        }
    }
}

# ══════════════════════════════════════════
# PHASE C: Kill ALL non-essential scheduled tasks
# ══════════════════════════════════════════
Write-Host "`n--- Phase C: Disabling scheduled tasks temporarily ---" -ForegroundColor Cyan

$tasksToDisable = @(
    "\Microsoft\Windows\Application Experience\Microsoft Compatibility Appraiser",
    "\Microsoft\Windows\Application Experience\ProgramDataUpdater",
    "\Microsoft\Windows\Customer Experience Improvement Program\Consolidator",
    "\Microsoft\Windows\Customer Experience Improvement Program\UsbCeip",
    "\Microsoft\Windows\DiskDiagnostic\Microsoft-Windows-DiskDiagnosticDataCollector",
    "\Microsoft\Windows\Maps\MapsUpdateTask",
    "\Microsoft\Windows\Windows Error Reporting\QueueReporting"
)

foreach ($task in $tasksToDisable) {
    try {
        Disable-ScheduledTask -TaskPath (Split-Path $task) -TaskName (Split-Path $task -Leaf) -ErrorAction Stop | Out-Null
        Write-Host "  [DISABLED] $(Split-Path $task -Leaf)" -ForegroundColor Green
    } catch {
        Write-Host "  [SKIPPED] $(Split-Path $task -Leaf)" -ForegroundColor DarkGray
    }
}

# ══════════════════════════════════════════
# PHASE D: Kill all listening TCP connections
#          (except system ports)
# ══════════════════════════════════════════
Write-Host "`n--- Phase D: Closing non-system listening ports ---" -ForegroundColor Cyan

$systemPorts = @(135,445,139,49664,49665,49666,49667,49668,5357,5358,1900)
$portKilled = 0

Get-NetTCPConnection -State Listen -ErrorAction SilentlyContinue | ForEach-Object {
    if ($systemPorts -notcontains $_.LocalPort) {
        $proc = Get-Process -Id $_.OwningProcess -ErrorAction SilentlyContinue
        if ($proc -and $protected -notcontains $proc.ProcessName) {
            Write-Host "  [PORT] Closing port $($_.LocalPort) (PID: $($_.OwningProcess), $($proc.ProcessName))" -ForegroundColor Yellow
            Stop-Process -Id $_.OwningProcess -Force -ErrorAction SilentlyContinue
            $portKilled++
        }
    }
}
Write-Host "  Closed $portKilled listening connections" -ForegroundColor Green

# ══════════════════════════════════════════
# PHASE E: CMD taskkill — final sweep
# ══════════════════════════════════════════
Write-Host "`n--- Phase E: CMD taskkill sweep ---" -ForegroundColor Cyan

$protectedNames = @(
    "explorer.exe","obs64.exe","obs32.exe","MsMpEng.exe",
    "Memory Compression","System","dwm.exe","csrss.exe",
    "lsass.exe","services.exe","svchost.exe","wininit.exe",
    "smss.exe","fontdrvhost.exe","sihost.exe","ctfmon.exe",
    "RuntimeBroker.exe","ShellExperienceHost.exe",
    "StartMenuExperienceHost.exe","SearchHost.exe",
    "TextInputHost.exe","taskhostw.exe","conhost.exe",
    "audiodg.exe","spoolsv.exe","SecurityHealthService.exe",
    "SecurityHealthSystray.exe","Sppsvc.exe","WmiPrvSE.exe",
    "powershell.exe","pwsh.exe","WindowsTerminal.exe"
)

$filterStrings = @('/fi "USERNAME eq ' + $env:USERNAME + '"')
foreach ($p in $protectedNames) {
    $filterStrings += '/fi "IMAGENAME ne ' + $p + '"'
}
$cmdArgs = "taskkill /f " + ($filterStrings -join " ")
Write-Host "  Running: $cmdArgs" -ForegroundColor DarkGray
cmd /c $cmdArgs 2>$null

# ══════════════════════════════════════════
# PHASE F: Kill all UDP listeners (non-system)
# ══════════════════════════════════════════
Write-Host "`n--- Phase F: Closing non-system UDP listeners ---" -ForegroundColor Cyan

Get-NetUDPEndpoint -ErrorAction SilentlyContinue | ForEach-Object {
    $proc = Get-Process -Id $_.OwningProcess -ErrorAction SilentlyContinue
    if ($proc -and $protected -notcontains $proc.ProcessName -and $_.LocalPort -gt 1024) {
        Write-Host "  [UDP] Port $($_.LocalPort) (PID: $($_.OwningProcess), $($proc.ProcessName))" -ForegroundColor Yellow
        Stop-Process -Id $_.OwningProcess -Force -ErrorAction SilentlyContinue
    }
}

# ══════════════════════════════════════════
# PHASE G: Clear all temp files
# ══════════════════════════════════════════
Write-Host "`n--- Phase G: Clearing temp files ---" -ForegroundColor Cyan

$tempPaths = @(
    "$env:TEMP",
    "C:\Windows\Temp",
    "$env:LOCALAPPDATA\Temp"
)

$totalCleanedMB = 0
foreach ($tp in $tempPaths) {
    if (Test-Path $tp) {
        $files = Get-ChildItem $tp -Recurse -ErrorAction SilentlyContinue
        $sizeMB = [math]::Round(($files | Measure-Object Length -Sum).Sum / 1MB, 0)
        Remove-Item "$tp\*" -Recurse -Force -ErrorAction SilentlyContinue
        $totalCleanedMB += $sizeMB
        Write-Host "  [CLEANED] $tp (~$sizeMB MB)" -ForegroundColor Green
    }
}

# ══════════════════════════════════════════
# PHASE H: Flush DNS and clear network cache
# ══════════════════════════════════════════
Write-Host "`n--- Phase H: Flushing network caches ---" -ForegroundColor Cyan
ipconfig /flushdns | Out-Null
Write-Host "  [FLUSHED] DNS cache" -ForegroundColor Green

# ══════════════════════════════════════════
# PHASE I: Force garbage collection
# ══════════════════════════════════════════
Write-Host "`n--- Phase I: Clearing memory ---" -ForegroundColor Cyan
[System.GC]::Collect()
[System.GC]::WaitForPendingFinalizers()
[System.GC]::Collect()
Start-Sleep -Seconds 5

# ══════════════════════════════════════════
# PHASE J: RESULTS
# ══════════════════════════════════════════
$after = [math]::Round((Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory / 1MB, 2)
$freed = [math]::Round($after - $before, 2)

Write-Host "`n==========================================" -ForegroundColor Cyan
Write-Host "  RAM RESULTS" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "  BEFORE:   $before GB" -ForegroundColor Yellow
Write-Host "  AFTER:    $after GB" -ForegroundColor Green
Write-Host "  FREED:    $freed GB" -ForegroundColor Green
Write-Host "  TEMP:     ~$totalCleanedMB MB cleaned" -ForegroundColor Green
Write-Host "  PROCS:    $killedCount processes killed" -ForegroundColor Green
Write-Host "  PORTS:    $portKilled ports closed" -ForegroundColor Green
Write-Host "==========================================" -ForegroundColor Cyan

If ($after -ge 8) {
    Write-Host "  [READY]  $after GB free — Safe for OBS + RDP + Browser" -ForegroundColor Green
} ElseIf ($after -ge 6) {
    Write-Host "  [OK]     $after GB free — Can start, close extra browser tabs" -ForegroundColor Yellow
} Else {
    Write-Host "  [LOW]    $after GB free — Restart laptop, open only what's needed" -ForegroundColor Red
}

# Show what's still running
Write-Host "`n--- Remaining processes (top 15 by RAM) ---" -ForegroundColor Cyan
Get-Process | Sort-Object WorkingSet64 -Descending |
    Select-Object -First 15 Name,
    @{N='RAM_MB';E={[math]::Round($_.WorkingSet64/1MB,0)}},
    Id | Format-Table -AutoSize

# Show remaining listening ports
Write-Host "--- Remaining listening ports ---" -ForegroundColor Cyan
Get-NetTCPConnection -State Listen -ErrorAction SilentlyContinue |
    Select-Object LocalPort, OwningProcess,
    @{N='Process';E={(Get-Process -Id $_.OwningProcess -ErrorAction SilentlyContinue).ProcessName}} |
    Sort-Object LocalPort | Format-Table -AutoSize
```

### Error Handling:

| Error / Situation | Cause | Fix |
|---|---|---|
| `Access denied` on Stop-Process | Process is system-protected | **Ignored automatically** — script moves to next process |
| `Service cannot be stopped` | Windows protects certain services | **Ignored automatically** — script prints [SKIPPED] |
| `Get-NetTCPConnection — Access denied` | PowerShell not running as Admin | **Close PowerShell → Reopen as Admin (Win + X → Terminal Admin)** |
| `Disable-ScheduledTask — not found` | Task doesn't exist on this system | **Ignored automatically** — script prints [SKIPPED] |
| `taskkill — no process found` | All user processes already killed in Phase A | **Normal** — Phase E found nothing extra to kill |
| `Remove-Item — file in use` | Temp files locked by running processes | **Ignored automatically** — moves to next file |
| Free RAM still below 6 GB | System processes consuming too much | **Restart laptop → Don't open any apps → Run Step 1 again** |
| Free RAM DECREASES after script | Services auto-restarted by Windows | **Run script one more time** — some services restart once |
| Screen goes blank briefly | Explorer restarted or desktop refreshed | **Normal** — desktop will recover in 2–3 seconds |
| Script shows syntax error on paste | Invisible characters from copy-paste | **Delete everything in PowerShell → Re-copy entire script from document → Paste again** |
| `term is not recognized as cmdlet` | Incomplete paste or line break issue | **Ensure you copied from `Write-Host` at the very top to the very last `Format-Table` at the bottom** |
| Taskbar disappears | explorer.exe was accidentally killed | **Press `Ctrl + Shift + Esc` → File → Run new task → Type `explorer.exe` → OK** |
| OBS closes during cleanup | OBS not in protected list | **This shouldn't happen** — obs64 and obs-studio are in the protected list. If it does: reopen OBS |
| Antivirus warning | Script is killing many processes rapidly | **Click Allow / Trust** — the script is safe, it only kills user processes |

## Step 2 — Check System Info

```powershell
$os = Get-CimInstance Win32_OperatingSystem
$cpu = Get-CimInstance Win32_Processor
$totalRAM = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory/1GB,2)
$freeRAM = [math]::Round($os.FreePhysicalMemory/1MB,2)
$diskFree = [math]::Round((Get-CimInstance Win32_LogicalDisk -Filter "DeviceID='C:'").FreeSpace/1GB,2)

Write-Host "======================================" -ForegroundColor Cyan
Write-Host "  SYSTEM INFO" -ForegroundColor Cyan
Write-Host "======================================" -ForegroundColor Cyan
Write-Host "OS:       $($os.Caption) $($os.OSArchitecture)"
Write-Host "CPU:      $($cpu.Name)"
Write-Host "Cores:    $($cpu.NumberOfCores)"
Write-Host "RAM:      $totalRAM GB"
Write-Host "Free RAM: $freeRAM GB"
Write-Host "Disk C:   $diskFree GB free"
Write-Host "======================================" -ForegroundColor Cyan

If($os.OSArchitecture -eq "64-bit"){Write-Host "[OS]      PASS" -ForegroundColor Green}else{Write-Host "[OS]      FAIL" -ForegroundColor Red}
If($cpu.NumberOfCores -ge 2){Write-Host "[CPU]     PASS" -ForegroundColor Green}else{Write-Host "[CPU]     FAIL" -ForegroundColor Red}
If($totalRAM -ge 8){Write-Host "[RAM]     PASS" -ForegroundColor Green}else{Write-Host "[RAM]     FAIL" -ForegroundColor Red}
If($freeRAM -ge 6){Write-Host "[FREE]    PASS ($freeRAM GB)" -ForegroundColor Green}else{Write-Host "[FREE]    LOW ($freeRAM GB)" -ForegroundColor Yellow}
If($diskFree -ge 10){Write-Host "[DISK]    PASS" -ForegroundColor Green}else{Write-Host "[DISK]    FAIL" -ForegroundColor Red}
```

### Error Handling:

| Result | Meaning | Action |
|---|---|---|
| **[OS] FAIL** | Not Windows 10/11 or not 64-bit | **Cannot proceed** — need Windows 10/11 64-bit machine |
| **[CPU] FAIL** | Less than 2 CPU cores | **Cannot proceed** — need a different machine |
| **[RAM] FAIL** | Total RAM below 8 GB | **Cannot proceed** — VM needs minimum 8 GB host RAM |
| **[FREE] LOW** | Free RAM below 6 GB | **Run Step 1 again** → If still low: **restart laptop → run Step 1 immediately before opening anything** |
| **[DISK] FAIL** | C: drive below 10 GB free | **Run:** `cleanmgr /d C:` → Check all boxes → OK → Or delete files from Downloads folder |
| All PASS | System meets requirements | **Proceed to Step 3** |

## Step 3 — Check Network

```powershell
Write-Host "======================================" -ForegroundColor Cyan
Write-Host "  NETWORK CHECK" -ForegroundColor Cyan
Write-Host "======================================" -ForegroundColor Cyan

try {
    $g = Test-Connection google.com -Count 2 -Quiet
    If ($g) { Write-Host "[INTERNET] PASS" -ForegroundColor Green }
    else { Write-Host "[INTERNET] FAIL" -ForegroundColor Red }
} catch {
    Write-Host "[INTERNET] FAIL — $($_.Exception.Message)" -ForegroundColor Red
```
